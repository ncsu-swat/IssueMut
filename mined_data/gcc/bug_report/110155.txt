#include <stdbool.h>

_Bool foo (void);

int bar (int r)
{
  if (foo ())
    r++;

  return r;
}
--cut here--

compiles (gcc -O2) to:

        movl    %edi, %ebx
        call    foo
        cmpb    $1, %al
        sbbl    $-1, %ebx
        movl    %ebx, %eax

could be performed without compare as:

        movl    %edi, %ebx
        callq   foo
        movzbl  %al, %eax
        addl    %ebx, %eax
