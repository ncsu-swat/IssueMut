- User: Xi Ruoyao | Time: 2023-08-31 04:31:18 UTC
  Text: int test(int a, int b)
{
  return (a & ~0x10) | (b & 0x10);
}

compiles to:

	addi.w	$r12,$r0,-17			# 0xffffffffffffffef
	and	$r12,$r12,$r4
	andi	$r5,$r5,16
	or	$r12,$r12,$r5
	slli.w	$r4,$r12,0
	jr	$r1

It should be improved:

bstrpick.w $r4, $r4, 4, 4
bstrins.w  $r5, $r4, 4, 4
or         $r5, $r4, $r0
