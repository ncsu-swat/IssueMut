typedef short int16_t;

#define QUANT_ONE( coef, mf, f )                \
{ \
    if( (coef) > 0 ) \
        (coef) = (f + (coef)) * (mf) >> 16; \
    else \
        (coef) = - ((f - (coef)) * (mf) >> 16); \
    nz |= (coef); \
}

int quant_4x4( int16_t dct[16], uint16_t mf[16], uint16_t bias[16] )
{
    int nz = 0;
    for( int i = 0; i < 16; i++ )
        QUANT_ONE( dct[i], mf[i], bias[i] );
    return !!nz;
}

    Simplify (AND (ASHIFTRT A imm) mask) to (LSHIFTRT A imm) for vector mode.
    
    When mask is (1 << (prec - imm) - 1) which is used to clear upper bits
    of A, then it can be simplified to LSHIFTRT.
    
    i.e Simplify
    (and:v8hi
      (ashifrt:v8hi A 8)
      (const_vector 0xff x8))
    to
    (lshifrt:v8hi A 8)
