/* file: f.cpp */
extern double a1; // 1.0
extern double a2; // 1.0

void f(double K[2], bool b)
{
    double A[] = {
        b ? a1 : a2,
        0,
        0,
        0
    };

    double sum{};
    for(double  a : A) sum += a;
    for(double& a : A) a /= sum;

    if (b) {
        K[0] = A[0]; // 1.0
        K[1] = A[1]; // 0.0
    } else {
        K[0] = A[0] + A[1];
    }
}
