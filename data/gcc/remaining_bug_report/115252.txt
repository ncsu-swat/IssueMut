void pixel_sub_wxh(int16_t *diff, uint8_t *pix1, uint8_t *pix2) {

void pixel_sub_wxh_vec(int16_t *diff, uint8_t *pix1, uint8_t *pix2) {
  for (int y = 0; y < 4; y++) {
