{
  f(1, 2, 3, 4);
}

fail:

        movs    r3, #4
        movs    r2, #3
        movs    r1, #2
        movs    r0, #1
        push    {ip, lr}
        pop     {ip, lr}
        aut     ip, lr, sp
        bx      ip  @ indirect register sibling call
