We can use 128-bit dot_prod instruction + clean upper 64 bits. Currently, gcc generates a long instruction sequence.

        vmovq   xmm0, QWORD PTR [rsi]
        vmovq   xmm2, QWORD PTR [rdx]
        vpmovzxbw       xmm1, xmm0
        vpsrlq  xmm0, xmm0, 32
        vpmovsxbw       xmm3, xmm2
        vpmullw xmm1, xmm1, xmm3
        vpsrlq  xmm2, xmm2, 32
        vpmovzxbw       xmm0, xmm0
        vpmovsxbw       xmm2, xmm2
        vpmullw xmm0, xmm0, xmm2
        vpmovsxwd       xmm2, xmm1
        vpsrlq  xmm1, xmm1, 32
        vpmovsxwd       xmm1, xmm1
        vpaddd  xmm2, xmm2, xmm1
        vpmovsxwd       xmm1, xmm0
        vpsrlq  xmm0, xmm0, 32
        vpmovsxwd       xmm0, xmm0
        vpaddd  xmm1, xmm1, xmm2
