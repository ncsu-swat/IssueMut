The following code (testcase.c) causes an ICE when using RISC-V vector as target.

a;
b() {
  long *c = 0;
  int *d;
  for (; a; ++a)
    c[a] = d[-a];
}

riscv64-unknown-linux-gnu-gcc -march=rv64gcv -mabi=lp64d \
